cmake_minimum_required(VERSION 3.21)
project(DeskNestPogget LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(desknest_pogget SHARED bridge.cpp ../pogget/PoggetCoreManager.cpp)
target_compile_definitions(desknest_pogget PRIVATE DESKNEST_POGGET_EXPORT)
target_include_directories(desknest_pogget PUBLIC .)
enable_testing()
add_executable(native_tests ../../tests/Native.Tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE desknest_pogget)
add_test(NAME native_tests COMMAND native_tests)
